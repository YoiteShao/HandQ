# Async HTTP client. Used directly by AnthropicStreamingService and

# OpenAI-compatible client used by VisionClient against the QGenie gateway

# vision fallback.
openai>=1.30.0

# ── Configuration ─────────────────────────────────────────────────────────────

# YAML configuration loader (src/infrastructure/config_manager.py,

pyyaml>=6.0.1

# ── Utilities ─────────────────────────────────────────────────────────────────

# JSON repair for truncated/malformed LLM responses (src/infrastructure/utils.py)
json-repair>=0.30.0

# ── SSH tool dependencies ──────────────────────────────────────────────────────

# SSH client (src/tools/ssh_tool.py, src/infrastructure/ssh_setup.py)
paramiko>=3.0.0

# OS keyring integration for SSH password storage. Stores credentials in
# Windows Credential Manager / Linux Secret Service / macOS Keychain.
keyring>=24.0.0

# Fallback file-based keyring backend for headless / Linux environments. Used
# automatically when the system keyring (D-Bus / Secret Service) is unavailable.
keyrings.alt>=4.0.0

# ── Browser tool ──────────────────────────────────────────────────────────────

# Playwright drives the browser tool. Edge / Chrome are detected at runtime;
# bundled Chromium is a fallback that requires `playwright install chromium`.
# The agent runs without this dependency — BrowserTool reports an actionable
# install hint when imports fail.
playwright>=1.40.0

# ── Vision (find_element + browser canvas) ────────────────────────────────────

# Image resize / format normalisation before vision calls. Used by
# infrastructure/vision/llm.py and desktop_tool's screenshot pipeline.
pillow>=10.0.0

# ── Desktop tool dependencies (Windows-only) ──────────────────────────────────
# desktop_tool degrades to a clear "install X" error when these are missing,
# so the agent runs without them — they're only needed when the Planner
# activates the desktop tool.

# Cross-platform fast screen capture (~5ms/frame on 1080p). Used by both
# desktop_tool and the Phase 3 activity_monitor.
mss>=9.0.0

# Mouse / keyboard automation. desktop_tool sets pyautogui.FAILSAFE=True so a
# panicked user can abort by ramming the mouse into a screen corner. Process-
# wide Ctrl+Shift+C revoke from the Electron overlay is the cleaner exit.
pyautogui>=0.9.54

# Win32 API bindings — desktop_tool uses win32gui for foreground-window
# inspection + EnumWindows, win32process for PID lookups. Imported lazily;
# missing pywin32 reduces list_windows / sensitive-window detection but
# leaves screenshot + input working.
pywin32>=306; sys_platform == 'win32'

# Local OCR engine selected by Phase 0 benchmark. Used by desktop_tool
# find_element (primary path) and Phase 3 activity_monitor.
# Note: 1.4.x caps at Python <3.13; requires Python 3.12 or below.
rapidocr-onnxruntime>=1.3.0

# Fuzzy string matching for find_element's OCR-text → description match.
# Falls back to substring match when missing — but install for proper
# token_set_ratio + partial_ratio scoring.
rapidfuzz>=3.0.0

# Optional. desktop_tool uses psutil to resolve process names from window PIDs
# (list_windows + sensitive-window guard); without it those fields are empty
# but the tool still works. Imported lazily inside try/except so the agent
# runs without it.
psutil>=5.9.0

# ── Teams tool dependencies (Windows-only) ────────────────────────────────────
# teams_tool talks to Microsoft Graph REST API. Auth is bootstrapped from the
# user's already-signed-in Teams Web session (read via Playwright against the
# shared HandQ browser_profile), so we don't need msal here. httpx (already
# above) handles the OAuth refresh-token exchange and Graph requests; playwright
# (already above for browser_tool) handles the one-time bootstrap.
