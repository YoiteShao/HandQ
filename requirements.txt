# the gap at startup rather than at first browser action.

# Image resize / format normalisation before vision calls. Used by

# ── Desktop tool dependencies (Windows-only) ──────────────────────────────────
# All listed deps are imported eagerly at desktop_tool module load. The tool is
# registered Windows-only in ToolRegistry, so Linux runs simply skip the import
# path entirely.

# Cross-platform fast screen capture (~5ms/frame on 1080p). Used by both
# desktop_tool and the Phase 3 activity_monitor.
mss>=9.0.0

# Mouse / keyboard automation. desktop_tool sets pyautogui.FAILSAFE=True so a
# panicked user can abort by ramming the mouse into a screen corner. Process-
# wide Ctrl+Shift+C revoke from the Electron overlay is the cleaner exit.
pyautogui>=0.9.54

# Win32 API bindings — desktop_tool uses win32gui for foreground-window
# inspection + EnumWindows, win32process for PID lookups. Hard dep on Windows.
pywin32>=306; sys_platform == 'win32'

# UI Automation client — desktop_tool's primary path for pointing at controls
# discovered via UIA tree walks (descendants/from_point).
pywinauto>=0.6.8; sys_platform == 'win32'

# Local OCR engine selected by Phase 0 benchmark. Used by desktop_tool
# find_element (primary path) and Phase 3 activity_monitor.
# Note: 1.4.x caps at Python <3.13; requires Python 3.12 or below.
rapidocr-onnxruntime>=1.3.0

# Fuzzy string matching for find_element's OCR-text → description match.
# Eager dep — used unconditionally for token_set_ratio + partial_ratio scoring.
rapidfuzz>=3.0.0

# Process / window-PID resolution. Used by desktop_tool list_windows and the
# personality monitor's input-idle window owner. Eager top-level import.
psutil>=5.9.0
